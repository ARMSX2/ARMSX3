# The SPU object cache key: a hash of every input that can change SPU machine code, so an app update keeps the cache.
set(ARMSX3_SPU_CODEGEN_SOURCES
    rpcs3/Emu/Cell/SPULLVMRecompiler.cpp
    rpcs3/Emu/Cell/SPUCommonRecompiler.cpp
    rpcs3/Emu/Cell/SPURecompiler.h
    rpcs3/Emu/Cell/SPUThread.h
    rpcs3/Emu/CPU/CPUThread.h
    rpcs3/Emu/CPU/CPUTranslator.h
    rpcs3/Emu/CPU/CPUTranslator.cpp
    Utilities/JITLLVM.cpp
    Utilities/JIT.h
    ios/core/vm_reservation_slot.h
    rpcs3/Emu/Cell/SPUOpcodes.h
    rpcs3/Emu/Cell/SPUAnalyser.h
    # The GHC frame pass rewrites every SPU function's prologue and returns as inline asm
    rpcs3/Emu/CPU/Backends/AArch64/AArch64JIT.cpp
    rpcs3/Emu/CPU/Backends/AArch64/AArch64JIT.h
    rpcs3/Emu/CPU/Backends/AArch64/AArch64ASM.cpp
    rpcs3/Emu/CPU/Backends/AArch64/AArch64ASM.h
    rpcs3/Emu/CPU/Backends/AArch64/AArch64Common.h
    # spu_mfc_cmd sits inside spu_thread, whose offsets the emitted code bakes in
    rpcs3/Emu/Cell/MFC.h
)
set(_armsx3_spu_identity_input "")
foreach(_armsx3_path IN LISTS ARMSX3_SPU_CODEGEN_SOURCES)
    file(SHA1 "${CMAKE_SOURCE_DIR}/${_armsx3_path}" _armsx3_sha)
    # Repo-relative paths, so two checkouts at different locations agree.
    string(APPEND _armsx3_spu_identity_input "${_armsx3_path}=${_armsx3_sha};")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/${_armsx3_path}")
endforeach()
# Only op_branch_targets is hashed from SPUThread.cpp, so the file's other edits keep the cache.
file(READ "${CMAKE_SOURCE_DIR}/rpcs3/Emu/Cell/SPUThread.cpp" _armsx3_spu_thread_text)
string(FIND "${_armsx3_spu_thread_text}" "\nstd::array<u32, 2> op_branch_targets(u32 pc, spu_opcode_t op)\n{\n" _armsx3_obt_begin)
if(_armsx3_obt_begin EQUAL -1)
    message(FATAL_ERROR "no op_branch_targets definition in rpcs3/Emu/Cell/SPUThread.cpp")
endif()
math(EXPR _armsx3_obt_begin "${_armsx3_obt_begin} + 1")
string(SUBSTRING "${_armsx3_spu_thread_text}" ${_armsx3_obt_begin} -1 _armsx3_obt_text)
string(FIND "${_armsx3_obt_text}" "\n}\n" _armsx3_obt_end)
if(_armsx3_obt_end EQUAL -1)
    message(FATAL_ERROR "op_branch_targets in rpcs3/Emu/Cell/SPUThread.cpp has no closing brace at column 0")
endif()
math(EXPR _armsx3_obt_end "${_armsx3_obt_end} + 2")
string(SUBSTRING "${_armsx3_obt_text}" 0 ${_armsx3_obt_end} _armsx3_obt_text)
string(SHA1 _armsx3_obt_sha "${_armsx3_obt_text}")
string(APPEND _armsx3_spu_identity_input "rpcs3/Emu/Cell/SPUThread.cpp#op_branch_targets=${_armsx3_obt_sha};")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/rpcs3/Emu/Cell/SPUThread.cpp")
# The superproject's LLVM gitlink; rev-parse inside an uninitialised submodule would return the superproject's HEAD.
execute_process(
    COMMAND git ls-tree HEAD 3rdparty/llvm/llvm
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE _armsx3_llvm_gitlink
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
)
string(APPEND _armsx3_spu_identity_input "llvm=${_armsx3_llvm_gitlink}")
string(SHA1 ARMSX3_SPU_CODEGEN_IDENTITY "${_armsx3_spu_identity_input}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/spu_codegen_identity.h" CONTENT [=[
// generated by ios/core/CMakeLists.txt, do not edit
#pragma once
#define ARMSX3_SPU_CODEGEN_IDENTITY "@ARMSX3_SPU_CODEGEN_IDENTITY@"
]=] @ONLY)
# A per-source include directory, so no other translation unit's command line changes.
set_property(SOURCE "${CMAKE_SOURCE_DIR}/rpcs3/Emu/Cell/SPUCommonRecompiler.cpp"
    TARGET_DIRECTORY rpcs3_emu APPEND PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}")
